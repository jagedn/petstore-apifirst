== CodeFirst

.PetStoreController.java
[source,java]
----
@Controller("/pets")
@Tag(name = "Pets", description = "Operaciones sobre el catálogo de mascotas")
public class PetstoreController {
    @Get("/{id}")
    @Produces(MediaType.APPLICATION_JSON)
    @Operation ...
    @ApiResponse(responseCode = "200" ...
    @ApiResponse(responseCode = "400" ...
    public HttpResponse<Pet> get(
            @Parameter(description = "Identificador único de la mascota", required = true)
            @PathVariable("id")
            @NotNull
            @Positive Long id
    ) {}
}
----

== ApiFirst, Paths

[.column]
[source,yaml]
----
paths:
  /v1/pets/{id}:
    get:
      parameters:
        - name: id
          in: path
          required: true
          schema:
            type: integer
            format: int64
            minimum: 1
----

[.column]
[source,yaml]
----
    (continua)
      responses:
        '200':
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/PetResponse'
        '404':
          description: Mascota no encontrada
----

== ApiFirst, Model

[.column]
[source,yaml]
----
components:
  schemas:
    CreatePetRequest:
      type: object
      required:
        - name
        - age
----

[.column]
[source,yaml]
----
    (continua)
    properties:
        name:
          type: string
          minLength: 2
          example: "Fido"
        age:
          type: integer
          format: int32
          minimum: 0
          example: 3
----
